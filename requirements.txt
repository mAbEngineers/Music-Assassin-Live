# macOS: use Python 3.11+ (3.12 recommended) with Tkinter. The current
# onnxruntime macOS wheel is arm64-only and needs macOS 14+; the Mac routing
# backend itself needs macOS 14.4+ for Core Audio process taps.
numpy
sounddevice
soxr
onnxruntime

# Windows only: the routing backend needs Core Audio endpoint enumeration
# (pycaw) and the COM plumbing to drive IPolicyConfig (comtypes). Marked by
# platform so a Linux install stays exactly as small as it was -- neither
# package has any use on Linux, and pycaw does not install there.
pycaw; sys_platform == "win32"
comtypes; sys_platform == "win32"
